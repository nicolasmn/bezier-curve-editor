Try it in plain HTML — no framework, no build step:

html
<script type="module">
  import '@bezier-curve-editor/core'
</script>

<bezier-curve-editor
  value="cubic-bezier(0.34, 1.56, 0.64, 1)"
></bezier-curve-editor>
